<script>
const BANK = [
  {q:"What is the capital of India?", a:["Mumbai","New Delhi","Kolkata","Chennai"], c:1},
  {q:"Which planet is known as the Red Planet?", a:["Earth","Mars","Jupiter","Venus"], c:1},
  {q:"Who wrote 'Hamlet'?", a:["Charles Dickens","William Shakespeare","Leo Tolstoy","Mark Twain"], c:1},
  {q:"Which gas do plants absorb from the atmosphere?", a:["Oxygen","Nitrogen","Carbon Dioxide","Hydrogen"], c:2},
  {q:"Which is the largest ocean on Earth?", a:["Atlantic","Indian","Pacific","Arctic"], c:2},
  {q:"Who is known as the father of computers?", a:["Charles Babbage","Alan Turing","Bill Gates","Steve Jobs"], c:0},
  {q:"What is the national animal of India?", a:["Peacock","Tiger","Elephant","Lion"], c:1},
  {q:"How many continents are there on Earth?", a:["5","6","7","8"], c:2},
  {q:"Which is the smallest planet?", a:["Mercury","Venus","Mars","Pluto"], c:0},
  {q:"What is H2O commonly known as?", a:["Salt","Water","Hydrogen","Oxygen"], c:1}
];

let profileName = localStorage.getItem('quiz_profile_name') || 'Gulshan';
let records = JSON.parse(localStorage.getItem('quiz_records') || '[]');
let settings = JSON.parse(localStorage.getItem('quiz_settings') || '{"numQuestions":10}');
document.getElementById('profileName').value = profileName;
document.getElementById('numQuestions').value = settings.numQuestions;
renderRecords();

function renderRecords(){
  const list = document.getElementById('recordList');
  list.innerHTML = records.length ? records.map(r=>`<div class='record-item'>${r.date} — ${r.name} — ${r.score}/${r.total}</div>`).join('') : '<div>No records yet</div>';
}

document.getElementById('openSettings').onclick=()=>drawer.classList.add('open');
document.getElementById('closeDrawer').onclick=()=>drawer.classList.remove('open');
document.getElementById('openSettingsBtn').onclick=()=>drawer.classList.add('open');
document.getElementById('saveSettings').onclick=()=>{
  profileName=document.getElementById('profileName').value||'Gulshan';
  settings.numQuestions=parseInt(document.getElementById('numQuestions').value);
  localStorage.setItem('quiz_profile_name',profileName);
  localStorage.setItem('quiz_settings',JSON.stringify(settings));
  alert('Saved!');
  drawer.classList.remove('open');
};
document.getElementById('clearRecords').onclick=()=>{
  if(confirm('Clear all records?')){records=[];localStorage.setItem('quiz_records','[]');renderRecords();}
};

let quiz=[],score=0,current=0,timer,timeLeft=10;
function shuffle(a){for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}}
function startGame(){
  score=0;current=0;
  quiz=[];
  for(let i=0;i<settings.numQuestions;i++){quiz.push(BANK[Math.floor(Math.random()*BANK.length)]);}
  document.querySelector('.card').style.display='none';
  document.getElementById('quizArea').style.display='block';
  document.getElementById('gameOverArea').style.display='none';
  updateScore();
  nextQ();
}
function nextQ(){
  if(current>=quiz.length){endGame();return;}
  const q=quiz[current];
  document.getElementById('questionText').textContent=`Q${current+1}. ${q.q}`;
  const box=document.getElementById('answersBox');box.innerHTML='';
  q.a.forEach((ans,i)=>{
    const b=document.createElement('button');
    b.className='answer-btn';b.textContent=ans;
    b.onclick=()=>check(i,q.c);
    box.appendChild(b);
  });
  timeLeft=10;timerEl.textContent=`Time left: ${timeLeft}s`;
  clearInterval(timer);
  timer=setInterval(()=>{timeLeft--;timerEl.textContent=`Time left: ${timeLeft}s`;if(timeLeft<=0){clearInterval(timer);current++;nextQ();}},1000);
}
function check(i,c){clearInterval(timer);if(i===c)score++;current++;updateScore();setTimeout(()=>nextQ(),200);}
function updateScore(){scoreEl.textContent=`Score: ${score}`;}
function endGame(){
  clearInterval(timer);
  document.getElementById('quizArea').style.display='none';
  document.getElementById('gameOverArea').style.display='block';
  document.getElementById('gameOverTitle').textContent=`Well done, ${profileName}!`;
  document.getElementById('finalScore').textContent=`Your score: ${score}/${quiz.length}`;
  const rec={date:new Date().toLocaleString(),name:profileName,score,total:quiz.length};
  records.push(rec);localStorage.setItem('quiz_records',JSON.stringify(records));renderRecords();
}
document.getElementById('startBtn').onclick=startGame;
document.getElementById('restartBtn').onclick=startGame;
document.getElementById('homeBtn').onclick=()=>{
  document.querySelector('.card').style.display='block';
  document.getElementById('quizArea').style.display='none';
  document.getElementById('gameOverArea').style.display='none';
};
